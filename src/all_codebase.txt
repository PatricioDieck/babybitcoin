// Internal helper: append a 32-byte hash in the form we store (LE bytes).
static void write_hash_le(Bytes& out, const Hash256& h) {
  out.insert(out.end(), h.begin(), h.end());
}

Bytes BlockHeader::serialize() const {
  Bytes out;
  out.reserve(80); // header is 80 bytes in Bitcoin
  write_u32_le(out, (uint32_t)version);
  write_hash_le(out, prev_block);
  write_hash_le(out, merkle_root);
  write_u32_le(out, timestamp);
  write_u32_le(out, bits);
  write_u32_le(out, nonce);
  return out;
}

Hash256 BlockHeader::hash() const {
  return dbl_sha256(serialize());
}

Bytes Block::serialize() const {
  Bytes out = header.serialize();
  write_varint(out, vtx.size());
  for (const auto& tx : vtx) {
    Bytes s = tx.serialize();
    out.insert(out.end(), s.begin(), s.end());
  }
  return out;
}

// Count leading zero bits by viewing the hash as big-endian bits for simplicity.
// Our Hash256 is stored LE by convention; reverse to BE view for bit counting.
static bool has_leading_zero_bits(const Hash256& h, uint32_t bits){
  // Convert to big-endian view of the 32-byte array
  Bytes be(h.rbegin(), h.rend()); // reverse the 32 bytes
  uint32_t full_zero_bytes = bits / 8;
  uint32_t extra_bits      = bits % 8;

  // Check full zero bytes
  for (uint32_t i=0;i<full_zero_bytes;i++){
    if (be[i] != 0) return false;
  }
  if (extra_bits == 0) return true;

  // Next byte must have its top 'extra_bits' zeroed.
  uint8_t mask = 0xFF << (8 - extra_bits); // e.g., extra_bits=3 => mask 0b1110_0000
  return (be[full_zero_bytes] & mask) == 0;
}

bool header_meets_pow(const BlockHeader& H) {
    return has_leading_zero_bits(H.hash(), H.bits);
  }


Hash256 merkle_root(const std::vector<Transaction>& vtx) {
  if (vtx.empty()) return Hash256{}; // all-zero if no txs (we'll avoid empty blocks in practice)

  // Start with txids (each already LE 32-byte hash)
  std::vector<Hash256> layer;
  layer.reserve(vtx.size());
  for (const auto& tx : vtx) layer.push_back(tx.txid());

  // While more than one hash remains, pairwise hash concatenations (dbl_sha256)
  while (layer.size() > 1) {
    std::vector<Hash256> next;
    next.reserve((layer.size()+1)/2);

    for (size_t i=0; i<layer.size(); i+=2) {
      const Hash256& a = layer[i];
      const Hash256& b = (i+1 < layer.size()) ? layer[i+1] : layer[i]; // duplicate last if odd

      Bytes cat;
      cat.insert(cat.end(), a.begin(), a.end());
      cat.insert(cat.end(), b.begin(), b.end());
      next.push_back(dbl_sha256(cat));
    }
    layer.swap(next);
  }
  return layer[0];
}

bool mine_block(Block& b){
  // Fill timestamp if caller didn't set it
  if (b.header.timestamp == 0) {
    b.header.timestamp = static_cast<uint32_t>(std::chrono::seconds(std::time(nullptr)).count());
  }
  // Recompute merkle root from current transactions
  b.header.merkle_root = merkle_root(b.vtx);

  // Simple nonce scan across 32-bit space
  for (uint32_t n = 0; n < 0xFFFFFFFFu; ++n) {
    b.header.nonce = n;
    Hash256 h = b.header.hash();
    if (has_leading_zero_bits(h, b.header.bits)) {
      return true; // found a valid PoW
    }
  }
  return false;
}

// A block header commits to: version, previous block, Merkle root of all txs,
// a timestamp, difficulty (we'll interpret 'bits' as a simple "leading zero
// bits"), and a nonce for PoW search.
//
// Whitepaper link: §3 (Timestamp Server) & §4 (Proof-of-Work)
//
// In full Bitcoin, 'bits' encodes a compact target; to keep happy-path minimal
// and readable, we use "number of required leading zero bits" (integer).
struct BlockHeader {
  int32_t version{1};
  Hash256 prev_block{};  // hash of previous block header (LE bytes)
  Hash256 merkle_root{}; // merkle root of txids (LE bytes)
  uint32_t timestamp{0}; // Unix epoch seconds
  uint32_t bits{0};      // simplified difficulty: "leading-zero bits" required
  uint32_t nonce{0};     // PoW search variable

  // Serialize header exactly in Bitcoin's order and endianness.
  Bytes serialize() const;

  // Double-SHA256 of the serialized header; this is the block header hash.
  Hash256 hash() const;
};

// A block is a header plus a list of transactions.
// The first transaction should be a coinbase (incentive), but we'll enforce
// that in the validation/chain module; here we just carry the data.
struct Block {
  BlockHeader header;
  std::vector<Transaction> vtx;

  // Full serialization: header + varint(tx count) + each tx serialization.
  Bytes serialize() const;
};

// Compute the Bitcoin-style merkle root over txids (duplicating the last hash

Hash256 merkle_root(const std::vector<Transaction> &vtx);

// Returns true if found a solution (header.nonce set), false otherwise.

// --- helpers ---

ChainState::UTXOKey ChainState::make_key(const Hash256& txid, uint32_t index) {
  // Binary key: 32B txid (LE) + 4B index (LE)
  std::string k; k.resize(36);
  std::memcpy(k.data(), txid.data(), 32);
  k[32] = (char)(index & 0xff);
  k[33] = (char)((index>>8) & 0xff);
  k[34] = (char)((index>>16)& 0xff);
  k[35] = (char)((index>>24)& 0xff);
  return k;
}

bool ChainState::have_utxo(const Hash256& txid, uint32_t index) const {
  return utxo_.find(make_key(txid, index)) != utxo_.end();
}

const UTXO* ChainState::get_utxo(const Hash256& txid, uint32_t index) const {
  auto it = utxo_.find(make_key(txid, index));
  if (it == utxo_.end()) return nullptr;
  return &it->second;
}

void ChainState::add_utxo(const Hash256& txid, uint32_t index, int64_t value, const Bytes& spk) {
  utxo_[make_key(txid, index)] = UTXO{ value, spk };
}

bool ChainState::spend_utxo(const Hash256& txid, uint32_t index) {
  return utxo_.erase(make_key(txid, index)) == 1;
}

ChainState::ChainState() {
  // Start with "no tip". In a full node we'd have a genesis block here.
  tip_hash_ = Hash256{}; // all zeros
  height_   = 0;
}

// --- TX Validation ---

bool ChainState::is_coinbase(const Transaction& tx) const {
  if (tx.vin.size() != 1) return false;
  const auto& in0 = tx.vin[0];
  // Coinbase has prevout hash = 0..0 and index = 0xffffffff
  bool zero_hash = true;
  for (auto b : in0.prevout.txid) { if (b != 0) { zero_hash = false; break; } }
  return zero_hash && (in0.prevout.index == 0xffffffff);
}

bool ChainState::validate_tx(const Transaction& tx, int64_t& fee) const {
  // Coinbase is allowed to have no inputs and arbitrary scriptSig; fees computed at block-level.
  if (is_coinbase(tx)) {
    // We'll check subsidy budget at block level; fee here is 0 for coinbase.
    fee = 0;
    return true;
  }

  // Regular TX: check all inputs exist and are authorized; values balance.
  int64_t sum_in = 0;
  for (size_t i = 0; i < tx.vin.size(); ++i) {
    const auto& in = tx.vin[i];
    const UTXO* u = get_utxo(in.prevout.txid, in.prevout.index);
    if (!u) return false; // double-spend or unknown input

    // P2PKH only (happy path)
    if (!verify_p2pkh_input(tx, i, u->scriptPubKey)) return false;

    sum_in += u->value;
    if (sum_in < 0) return false; // overflow guard
  }

  int64_t sum_out = 0;
  for (const auto& o : tx.vout) {
    if (o.value < 0) return false;
    sum_out += o.value;
    if (sum_out < 0) return false; // overflow guard
  }

  if (sum_in < sum_out) return false; // cannot create money
  fee = sum_in - sum_out;
  return true;
}

// --- Block Validation & Connection ---

bool ChainState::validate_block(const Block& b, int64_t& total_fees) const {
  // 1) Basic structure: at least 1 tx (coinbase)
  if (b.vtx.empty()) return false;
  if (!is_coinbase(b.vtx[0])) {
    std::cerr << "Block does not have a coinbase transaction" << "\n";
    return false;
  }

  // 2) PoW correct for the header we see
  if (!header_meets_pow(b.header)) {
    std::cerr << "VALIDATION ERROR: Block header does not meet PoW" << "\n";
    return false;
  }

  // 3) Merkle root matches txids
  if (merkle_root(b.vtx) != b.header.merkle_root) {
    std::cerr << "VALIDATION ERROR: Merkle root does not match txids" << "\n";
    return false;
  }

  // 4) Validate txs; accumulate fees from non-coinbase txs
  total_fees = 0;
  for (size_t i = 0; i < b.vtx.size(); ++i) {
    int64_t fee = 0;
    if (!validate_tx(b.vtx[i], fee)) {
      std::cerr << "VALIDATION ERROR: Transaction is invalid" << "\n";
      return false;
    }
    if (i != 0) total_fees += fee; // coinbase fee handled as subsidy budget
  }

  // 5) Coinbase value <= SUBSIDY + fees (no money printing)
  // Happy path: fixed SUBSIDY (no halvings yet).
  int64_t coinbase_out = 0;
  for (const auto& o : b.vtx[0].vout) {
    if (o.value < 0) return false;
    coinbase_out += o.value;
    if (coinbase_out < 0) return false;
  }
  if (coinbase_out > SUBSIDY + total_fees) return false;

  return true;
}

bool ChainState::connect_block(const Block& b, int64_t total_fees) {
  // Spend inputs of all non-coinbase txs; then add all outputs of every tx.
  // If any spend/add fails mid-way, we'd need to roll back; happy path assumes valid input.

  // Spend inputs (skip coinbase)
  for (size_t i = 1; i < b.vtx.size(); ++i) {
    const auto& tx = b.vtx[i];
    for (const auto& in : tx.vin) {
      if (!spend_utxo(in.prevout.txid, in.prevout.index)) return false;
    }
  }

  // Add outputs (including coinbase)
  for (size_t i = 0; i < b.vtx.size(); ++i) {
    const auto& tx = b.vtx[i];
    Hash256 id = tx.txid();
    for (uint32_t n = 0; n < tx.vout.size(); ++n) {
      const auto& o = tx.vout[n];
      add_utxo(id, n, o.value, o.scriptPubKey);
    }
  }

  // Advance tip (happy path: append only)
  tip_hash_ = b.header.hash();
  height_  += 1;
  (void)total_fees; // not used further yet
  return true;
}

bool ChainState::accept_block(const Block& b) {
  int64_t fees = 0;
  if (!validate_block(b, fees)) return false;
  return connect_block(b, fees);
}

// Balance helper: sum UTXOs that match HASH160(pub).
int64_t ChainState::balance_for_pubkey(const Bytes& pub) const {
  Bytes target = hash160(pub);
  int64_t sum = 0;
  for (const auto& [k, u] : utxo_) {
    // Try to parse a P2PKH and match h160
    try {
      Bytes h = p2pkh_hash_from_script(u.scriptPubKey);
      if (h == target) {
        sum += u.value;
        if (sum < 0) return 0; // overflow guard; shouldn't happen
      }
    } catch (...) {
      // Not a P2PKH; ignore (we only create P2PKH here anyway)
    }
  }
  return sum;
}

// Turn raw bytes into readable hex for debugging.
// We'll use this constantly to verify serialized txs/blocks against
// expectations.
std::string hex(const Bytes &b) {
  std::ostringstream oss;
  for (auto v : b)
    oss << std::hex << std::setw(2) << std::setfill('0') << (int)v;
  return oss.str();
}

// Helper to convert a single hex char to its numeric value.
static uint8_t hexval(char c) {
  if ('0' <= c && c <= '9')
    return c - '0';
  if ('a' <= c && c <= 'f')
    return c - 'a' + 10;
  if ('A' <= c && c <= 'F')
    return c - 'A' + 10;
  throw std::runtime_error("bad hex");
}

// Parse a hex string into bytes. Useful for tests and fixtures.
// Example: from_hex("0a0b") -> {0x0a, 0x0b}
Bytes from_hex(const std::string &s) {
  if (s.size() % 2)
    throw std::runtime_error("hex length");
  Bytes out;
  out.reserve(s.size() / 2);
  for (size_t i = 0; i < s.size(); i += 2)
    out.push_back((hexval(s[i]) << 4) | hexval(s[i + 1]));
  return out;
}

// Little-endian writers (Bitcoin's canonical integer encoding on wire/disk).
// These must be exact; changing byte order changes hashes.
void write_u32_le(Bytes &out, uint32_t v) {
  out.push_back((v >> 0) & 0xff);
  out.push_back((v >> 8) & 0xff);
  out.push_back((v >> 16) & 0xff);
  out.push_back((v >> 24) & 0xff);
}

void write_u64_le(Bytes &out, uint64_t v) {
  for (int i = 0; i < 8; i++)
    out.push_back((v >> (8 * i)) & 0xff);
}

void write_i64_le(Bytes &out, int64_t v) {
  write_u64_le(out, static_cast<uint64_t>(v));
}

// CompactSize (varint) used for counts/lengths in tx/block serialization.
// Spec: 1 byte if < 0xfd; 0xfd + 2 bytes if <= 0xffff; 0xfe + 4 bytes if <=
// 0xffffffff; else 0xff + 8 bytes.
void write_varint(Bytes &out, uint64_t v) {
  if (v < 0xfd) {
    out.push_back((uint8_t)v);
    return;
  }
  if (v <= 0xffff) {
    out.push_back(0xfd);
    out.push_back(v & 0xff);
    out.push_back((v >> 8) & 0xff);
    return;
  }
  if (v <= 0xffffffffULL) {
    out.push_back(0xfe);
    for (int i = 0; i < 4; i++)
      out.push_back((v >> (8 * i)) & 0xff);
    return;
  }
  out.push_back(0xff);
  for (int i = 0; i < 8; i++)
    out.push_back((v >> (8 * i)) & 0xff);
}

// Reader counterpart for when we parse payloads (e.g., from peers or disk).
uint64_t read_varint(const uint8_t *&p, const uint8_t *end) {
  if (p >= end)
    throw std::runtime_error("varint eof");
  uint8_t b = *p++;
  if (b < 0xfd)
    return b;
  if (b == 0xfd) {
    if (end - p < 2)
      throw std::runtime_error("eof");
    uint64_t v = p[0] | (uint64_t)p[1] << 8;
    p += 2;
    return v;
  }
  if (b == 0xfe) {
    if (end - p < 4)
      throw std::runtime_error("eof");
    uint64_t v = 0;
    for (int i = 0; i < 4; i++)
      v |= (uint64_t)p[i] << (8 * i);
    p += 4;
    return v;
  }
  if (end - p < 8)
    throw std::runtime_error("eof");
  uint64_t v = 0;
  for (int i = 0; i < 8; i++)
    v |= (uint64_t)p[i] << (8 * i);
  p += 8;
  return v;
}#pragma once

Hash256 sha256(const Bytes &m) {
  Hash256 out{};
  ::SHA256(m.data(), m.size(), out.data());
  return out;
}

Hash256 dbl_sha256(const Bytes& m){
  auto h1 = sha256(m);
  Bytes tmp(h1.begin(), h1.end());
  return sha256(tmp);
}

Bytes ripemd160(const Bytes& m){
  Bytes out(20);
  ::RIPEMD160(m.data(), m.size(), out.data());
  return out;
}

Bytes hash160(const Bytes& m){
  auto h = sha256(m);
  Bytes tmp(h.begin(), h.end());
  return ripemd160(tmp);
}

// --- secp256k1 helper ---

static Bytes compress_pubkey(EC_KEY* key){
  // Force compressed form
  EC_KEY_set_conv_form(key, POINT_CONVERSION_COMPRESSED);
  int len = i2o_ECPublicKey(key, nullptr);
  if (len<=0) throw std::runtime_error("pub len");
  Bytes out(len);
  unsigned char* p = out.data();
  if (i2o_ECPublicKey(key, &p) != len) throw std::runtime_error("pub ser");
  return out;
}

// --- Keys & ECDSA ---

KeyPair generate_keypair(){
  KeyPair kp;
  EC_KEY* key = EC_KEY_new_by_curve_name(NID_secp256k1);
  if(!key) throw std::runtime_error("ec new");
  // ensure compressed form on this key too
  EC_KEY_set_conv_form(key, POINT_CONVERSION_COMPRESSED);

  if(EC_KEY_generate_key(key)!=1){ EC_KEY_free(key); throw std::runtime_error("ec gen"); }

  const BIGNUM* priv_bn = EC_KEY_get0_private_key(key);
  Bytes priv(32,0);
  BN_bn2binpad(priv_bn, priv.data(), 32);
  kp.priv = std::move(priv);

  // serialize compressed public key
  kp.pub  = compress_pubkey(key);
  EC_KEY_free(key);
  return kp;
}

// Sign the hash of the entire transaction with the private key of the owner of the previous output
Bytes ecdsa_sign(const Hash256 &h, const Bytes &priv) {
  // Build EC_KEY from private scalar
  EC_KEY *key = EC_KEY_new_by_curve_name(NID_secp256k1);
  if (!key)
    throw std::runtime_error("ec new");
  BIGNUM *d = BN_bin2bn(priv.data(), priv.size(), nullptr);
  if (!d) {
    EC_KEY_free(key);
    throw std::runtime_error("bn");
  }

  EC_KEY_set_private_key(key, d);

  // Derive public Q = d*G (needed for OpenSSL to sign in some configs)
  const EC_GROUP *grp = EC_GROUP_new_by_curve_name(NID_secp256k1);
  EC_POINT *Q = EC_POINT_new(grp);
  EC_POINT_mul(grp, Q, d, nullptr, nullptr, nullptr);
  EC_KEY_set_group(key, grp);
  EC_KEY_set_public_key(key, Q);

  ECDSA_SIG *sig = ECDSA_do_sign(h.data(), h.size(), key);
  if (!sig) {
    EC_POINT_free(Q);
    EC_GROUP_free((EC_GROUP *)grp);
    BN_free(d);
    EC_KEY_free(key);
    throw std::runtime_error("sign");
  }

  // DER-encode signature (Bitcoin expects DER for legacy ECDSA)
  int derlen = i2d_ECDSA_SIG(sig, nullptr);
  Bytes der(derlen);
  unsigned char *p = der.data();
  i2d_ECDSA_SIG(sig, &p);

  ECDSA_SIG_free(sig);
  EC_POINT_free(Q);
  EC_GROUP_free((EC_GROUP *)grp);
  BN_free(d);
  EC_KEY_free(key);

  return der;
}

bool ecdsa_verify(const Hash256 &h, const Bytes &sig_der, const Bytes &pub) {
  EC_KEY *key = EC_KEY_new_by_curve_name(NID_secp256k1);
  if (!key)
    return false;

  const unsigned char *pp = pub.data();
  if (o2i_ECPublicKey(&key, &pp, pub.size()) == nullptr) {
    EC_KEY_free(key);
    return false;
  }

  const unsigned char *sp = sig_der.data();
  ECDSA_SIG *sig = d2i_ECDSA_SIG(nullptr, &sp, sig_der.size());
  if (!sig) {
    EC_KEY_free(key);
    return false;
  }

  int ok = ECDSA_do_verify(h.data(), h.size(), sig, key);
  ECDSA_SIG_free(sig);
  EC_KEY_free(key);
  return ok == 1;
}

// Minimal opcodes we reference (for readability)
static constexpr uint8_t OP_DUP = 0x76, OP_HASH160 = 0xa9, OP_EQUALVERIFY = 0x88, OP_CHECKSIG = 0xac;

Bytes p2pkh_script(const Bytes& pubkey){
    Bytes h160 = hash160(pubkey);
    Bytes s;
    s.push_back(OP_DUP);
    s.push_back(OP_HASH160);
    s.push_back(0x14);
    s.insert(s.end(), h160.begin(), h160.end());
    s.push_back(OP_EQUALVERIFY);
    s.push_back(OP_CHECKSIG);
    return s;
}

Bytes p2pkh_hash_from_script(const Bytes &spk) {
  if (spk.size() == 25 && spk[0] == OP_DUP && spk[1] == OP_HASH160 &&
      spk[2] == 0x14 && spk[23] == OP_EQUALVERIFY && spk[24] == OP_CHECKSIG) {
    return Bytes(spk.begin() + 3, spk.begin() + 23);
  }
  throw std::runtime_error("not P2PKH");
}

static void ser_outpoint(Bytes& out, const OutPoint& op){
  // txid is serialized LE on the wire; we already store LE bytes
  out.insert(out.end(), op.txid.begin(), op.txid.end());
  write_u32_le(out, op.index);
}
static void ser_txin(Bytes& out, const TxIn& in){
  ser_outpoint(out, in.prevout);
  write_varint(out, in.scriptSig.size());
  out.insert(out.end(), in.scriptSig.begin(), in.scriptSig.end());
  write_u32_le(out, in.sequence);
}
static void ser_txout(Bytes& out, const TxOut& o){
  write_i64_le(out, o.value);
  write_varint(out, o.scriptPubKey.size());
  out.insert(out.end(), o.scriptPubKey.begin(), o.scriptPubKey.end());
}

Bytes Transaction::serialize(bool) const {
  Bytes out;
  out.reserve(10 + vin.size()*41 + vout.size()*34);
  // version
  write_u32_le(out, (uint32_t)version);
  // vin
  write_varint(out, vin.size());
  for (auto& in : vin) ser_txin(out, in);
  // vout
  write_varint(out, vout.size());
  for (auto& o : vout) ser_txout(out, o);
  // locktime
  write_u32_le(out, locktime);
  return out;
}

Hash256 Transaction::txid() const {
  return dbl_sha256(serialize());
}

Bytes build_p2pkh_scriptSig(const Bytes& sig_der, const Bytes& pub) {
  Bytes s;
  // push sig+hashtype (DER || 0x01)
  Bytes sig_plus = sig_der; sig_plus.push_back(0x01); // SIGHASH_ALL
  s.push_back((uint8_t)sig_plus.size());
  s.insert(s.end(), sig_plus.begin(), sig_plus.end());
  // push pubkey
  s.push_back((uint8_t)pub.size());
  s.insert(s.end(), pub.begin(), pub.end());
  return s;
}

Hash256 sighash_all(const Transaction& tx, size_t i, const Bytes& prevout_scriptPubKey) {
  if (i >= tx.vin.size()) throw std::runtime_error("sighash index");
  // Serialize tx with all inputs' scriptSig empty, except i replaced with prevout_scriptPubKey
  Transaction t = tx; // copy
  for (size_t k=0;k<t.vin.size();++k) t.vin[k].scriptSig.clear();
  t.vin[i].scriptSig = prevout_scriptPubKey;

  Bytes ser = t.serialize();
  // append 4-byte SIGHASH type
  write_u32_le(ser, 1); // SIGHASH_ALL
  return dbl_sha256(ser);
}

bool verify_p2pkh_input(const Transaction& tx, size_t i, const Bytes& prevout_spk) {
  // parse scriptSig
  if (i >= tx.vin.size()) return false;
  const Bytes& ss = tx.vin[i].scriptSig;
  if (ss.size()<2) return false;
  size_t p=0;
  uint8_t siglen = ss[p++]; if (p+siglen>ss.size()) return false;
  Bytes sig_plus(ss.begin()+p, ss.begin()+p+siglen); p+=siglen;
  if (sig_plus.empty() || sig_plus.back()!=0x01) return false; // must be SIGHASH_ALL
  sig_plus.pop_back(); // DER only
  if (p>=ss.size()) return false;
  uint8_t publen = ss[p++]; if (p+publen>ss.size()) return false;
  Bytes pub(ss.begin()+p, ss.begin()+p+publen);

  // check pubkey hash matches output script
  Bytes h160 = p2pkh_hash_from_script(prevout_spk);
  if (hash160(pub) != h160) return false;

  // compute SIGHASH
  Hash256 h = sighash_all(tx, i, prevout_spk);
  return ecdsa_verify(h, sig_plus, pub);
}

struct OutPoint {
  Hash256 txid;
  uint32_t index;
};

// Input: what we’re spending + the proof (scriptSig) we provide.
struct TxIn {
  OutPoint prevout;
  Bytes scriptSig; // for P2PKH- unlocks/signs tx to new owner: <sig (DER) + 1-byte SIGHASH_ALL> <pubkey>
  uint32_t sequence{0xffffffff}; 
};

// Output: value + locking script (scriptPubKey).
struct TxOut {
  int64_t value{0};   // satoshis
  Bytes scriptPubKey; // P2PKH script (25 bytes)/ locks/assigns tx to new owner
};

// A transaction: version, inputs, outputs, locktime.
struct Transaction {
  int32_t version{1};
  std::vector<TxIn> vin;
  std::vector<TxOut> vout;
  uint32_t locktime{0};

  Bytes serialize(bool with_witness = false) const; // minimal: no witness
  Hash256 txid() const; // dbl_sha256(serialized bytes)
};

// Build scriptSig (unlocking, ownership-proving script) for P2PKH: <sig DER + 0x01 SIGHASH_ALL> <pubkey>
Bytes build_p2pkh_scriptSig(const Bytes& sig_der, const Bytes& pub);

// Compute the SIGHASH_ALL digest for input i (legacy): all scriptSigs empty,
// except input i replaced with its prevout's scriptPubKey, then append
// 0x01000000.
Hash256 sighash_all(const Transaction& tx, size_t input_index, const Bytes &prevout_scriptPubKey);

// Verify a P2PKH input against its referenced output's scriptPubKey.
bool verify_p2pkh_input(const Transaction& tx, size_t i, const Bytes& prevout_scriptPubKey);
